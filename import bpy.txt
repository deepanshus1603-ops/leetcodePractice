import bpy
import json
import os


RIG_NAME = "RIG-rain"

TARGET_CONTROLS = [
    # Right arm - working wave
    "FK-Upperarm_Parent.R",
    "FK-Upperarm.R",
    "FK-Forearm.R",
    "FK-Hand.R",

    # Left arm
    "FK-Upperarm_Parent.L",
    "FK-Upperarm.L",
    "FK-Forearm.L",
    "FK-Hand.L",

    # Smile
    "ACT-Lips_Corner.L",
    "ACT-Lips_Corner.R",
    "ACT-Cheek_Outer.L",
    "ACT-Cheek_Outer.R",

    # Blink
    "ACT-Eyelid_Upper.L",
    "ACT-Eyelid_Upper.R",
    "ACT-Eyelid_Lower.L",
    "ACT-Eyelid_Lower.R",
]


def vec(v):
    return [round(float(x), 6) for x in v]


state = {
    "blender_version": bpy.app.version_string,
    "blend_file": bpy.data.filepath,
    "frame_current": bpy.context.scene.frame_current,
    "frame_start": bpy.context.scene.frame_start,
    "frame_end": bpy.context.scene.frame_end,
    "rig_name": RIG_NAME,
    "rig_found": False,
    "action": None,
    "controls": {}
}


rig = bpy.data.objects.get(RIG_NAME)

if rig is None:
    state["error"] = f"Rig '{RIG_NAME}' was not found."

else:
    state["rig_found"] = True
    state["rig_type"] = rig.type

    if rig.animation_data and rig.animation_data.action:
        state["action"] = rig.animation_data.action.name

    for bone_name in TARGET_CONTROLS:

        pb = rig.pose.bones.get(bone_name)

        if pb is None:
            state["controls"][bone_name] = {
                "exists": False
            }
            continue

        constraints = []

        for c in pb.constraints:
            item = {
                "name": c.name,
                "type": c.type,
                "mute": bool(c.mute)
            }

            if hasattr(c, "influence"):
                item["influence"] = round(float(c.influence), 4)

            constraints.append(item)

        state["controls"][bone_name] = {
            "exists": True,
            "location": vec(pb.location),
            "rotation_mode": pb.rotation_mode,
            "rotation_euler": vec(pb.rotation_euler),
            "rotation_quaternion": vec(pb.rotation_quaternion),
            "scale": vec(pb.scale),
            "constraints": constraints
        }


output_file = os.path.join(
    bpy.path.abspath("//"),
    "scene_state.json"
)

with open(output_file, "w", encoding="utf-8") as f:
    json.dump(state, f, indent=2)


print()
print("============================================")
print("RAIN_INSPECT_OK")
print("============================================")
print("Rig found:", state["rig_found"])
print("Action:", state["action"])
print("Frame:", state["frame_current"])
print("Controls inspected:", len(state["controls"]))
print("Output:", output_file)
print("============================================")
print()
